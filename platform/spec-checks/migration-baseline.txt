# 迁移基线：已发布的迁移文件 sha256
# 格式：<相对路径> <sha256>
# 维护：每次发布后把新迁移追加到这里并提交；已在基线中的文件不允许再改（改了 M1 会失败）
#
# 口径：哈希按「行尾归一化为 LF 之后的内容」计算，与检出平台无关。
# MigrationGuardTest 会先做同样的归一化再比对，否则 Windows 检出（CRLF）与
# Linux CI 检出（LF）会得到两个哈希，导致"迁移被修改"在其中一个平台误报。
#
# 生成（在 platform/ 下执行）：
#   PowerShell :
#     Get-ChildItem -Recurse -Filter "V*.sql" |
#       Where-Object { $_.FullName -match "db\\migration" } |
#       ForEach-Object {
#         $t = (Get-Content $_.FullName -Raw) -replace "`r`n", "`n" -replace "`r", "`n"
#         $h = [System.Security.Cryptography.SHA256]::Create().ComputeHash(
#                [System.Text.Encoding]::UTF8.GetBytes($t))
#         "{0} {1}" -f $_.FullName.Replace((Get-Location).Path + '\','').Replace('\','/'),
#                     ([System.BitConverter]::ToString($h) -replace '-','').ToLower()
#       }
#   bash（逐个文件，保留路径）:
#     find . -path '*/db/migration/V*.sql' -type f | while read -r f; do
#       printf '%s %s\n' "${f#./}" "$(sed 's/\r$//' "$f" | sha256sum | cut -d' ' -f1)"
#     done
